857:     fun findByDocumentIds(level: ExpertIndexLevel, ids: List<String>): List<ExpertProfile> {
858:         val documentIds = ids.filter { it.isNotBlank() }.distinct()
859:         if (documentIds.isEmpty()) return emptyList()

861:         val index = expertIndexService.indexName(level)
862:         val requestBody = mapOf(
863:             "docs" to documentIds.map { mapOf("_index" to index, "_id" to it, "_source" to sourceFields()) }
864:         )
865: 
866:         val response = restTemplate.exchange(
867:             "${properties.baseUrl}/_mget",
868:             HttpMethod.POST,
869:             HttpEntity(requestBody, headers()),
870:             JsonNode::class.java
871:         ).body ?: return emptyList()
872: 
873:         return response.path("docs")
874:             .filter { it.path("found").asBoolean(false) }
