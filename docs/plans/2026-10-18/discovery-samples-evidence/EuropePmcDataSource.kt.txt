72:     /** R-1（V-4）：XML 抓取结果 —— 区分「没取到内容」与「预算已尽、根本没发请求」，计数因此不会撒谎。 */
73:     private sealed class XmlFetchResult {
74:         class Bytes(val payload: ByteArray) : XmlFetchResult()
75:         class Failed(val requestsIssued: Int) : XmlFetchResult()
76:         class BudgetExhausted(val requestsIssued: Int) : XmlFetchResult()
77:     }
78: 
79:     /**
80:      * R-1（V-4）：XML 阶段在**同一个**单篇共享预算内完成 —— 已过期不发请求；真实请求的连接/读取超时
81:      * 取 `min(既有配置, 当时剩余预算)`，所以一个慢响应（连接、响应头或响应体）都只会被截断到剩余预算，
82:      * 不会把调用方拖过总时限；重试前同样重新判断预算，过期即停手（不再发多余请求）。
83:      */
84:     private fun fetchFullTextXml(pmcId: String, deadline: Instant?): XmlFetchResult {
85:         val url = "${properties.baseUrl}/$pmcId/fullTextXML"
86:         var requestsIssued = 0
87:         return try {
88:             if (properties.requestDelayMs > 0) {
89:                 val remaining = BoundedFulltextHttp.remainingMsOrUnbounded(deadline)
90:                 if (remaining != BoundedFulltextHttp.UNBOUNDED_REMAINING_MS && remaining <= properties.requestDelayMs) {
91:                     return XmlFetchResult.BudgetExhausted(requestsIssued)
92:                 }
93:                 Thread.sleep(properties.requestDelayMs)
94:             }
95:             val payload = FetchRetry.retryOnRecoverableIo<ByteArray?>(
96:                 maxRetries = properties.maxRetries,
97:                 initialBackoffMs = properties.retryBackoffMs
98:             ) {
99:                 val remaining = BoundedFulltextHttp.remainingMsOrUnbounded(deadline)
100:                 if (remaining <= 0L) return@retryOnRecoverableIo null
101:                 requestsIssued++
102:                 // R-1（V-4）：连接/读取超时取 min(既有配置, 剩余预算)，响应体也在同一个绝对 deadline
103:                 // 内读完 —— 细水长流的 XML 会被截断，而不是永远读下去。
104:                 BoundedFulltextHttp.getForObject(
105:                     restTemplate, url, ByteArray::class.java,
106:                     properties.connectTimeoutMs.toLong(), properties.readTimeoutMs.toLong(), deadline
107:                 )
108:             }
109:             if (payload == null) XmlFetchResult.BudgetExhausted(requestsIssued) else XmlFetchResult.Bytes(payload)
110:         } catch (e: Exception) {
111:             log.debug("Failed to fetch full text XML for {}: {}", pmcId, e.message)
112:             val expired = deadline != null && !Instant.now().isBefore(deadline)
113:             if (expired) {
114:                 XmlFetchResult.BudgetExhausted(requestsIssued)
115:             } else {
116:                 XmlFetchResult.Failed(requestsIssued.coerceAtLeast(1))
117:             }
118:         }
119:     }

138:     /**
139:      * R-4（V-4）：带共享剩余时限的 XML 阶段入口（OpenAlex 全文回退链传入单篇总时限）。
140:      *
141:      * 时限已过时**一个请求都不发**，并按既有的 `TIMEOUT` 类别上报（不是新的失败类别、也不是
142:      * `FULLTEXT_FETCH_FAILED`）—— 调用方据此停止后续 URL / Unpaywall 阶段。
143:      * 传 `null` 时行为与原来的 [extractAuthorEmails] 完全一致。
144:      */
145:     fun extractAuthorEmails(paper: PaperMetadata, deadline: Instant?): EmailExtractionOutcome {
146:         if (!properties.enabled) {
147:             return EmailExtractionOutcome(emptyList(), emailExtractionMethod, "SOURCE_DISABLED")
148:         }
149: 
150:         val searchEmails = paper.authors.mapNotNull { author ->
151:             val email = author.email?.trim()?.takeIf { it.isNotEmpty() } ?: return@mapNotNull null
152:             AuthorEmail(email, null, null, false, null, null)
153:         }
154:         // Search metadata alone remains an unverified clue; prefer explicit fulltext ownership.
155:         if (searchEmails.isNotEmpty() && paper.pmcId == null) {
156:             return EmailExtractionOutcome(searchEmails, "SEARCH_FIELD", null, httpRequests = 0)
157:         }
158: 
159:         val pmcId = paper.pmcId
160:         if (pmcId == null) {
161:             return EmailExtractionOutcome(emptyList(), emailExtractionMethod, "NO_PMC_ID")
162:         }
163: 
164:         if (deadline != null && !Instant.now().isBefore(deadline)) {
165:             log.debug("[{}] shared fulltext deadline expired before the XML stage for {}", sourceName, pmcId)
166:             return EmailExtractionOutcome(
167:                 emptyList(), emailExtractionMethod, "FULLTEXT_FETCH_FAILED",
168:                 httpRequests = 0, fulltextObtained = false,
169:                 downloadFailureCategory = FULLTEXT_FAILURE_TIMEOUT
170:             )
171:         }
172: 
173:         val xml = when (val fetched = fetchFullTextXml(pmcId, deadline)) {
174:             is XmlFetchResult.Bytes -> fetched.payload
175:             is XmlFetchResult.BudgetExhausted -> return EmailExtractionOutcome(
176:                 emptyList(), emailExtractionMethod, "FULLTEXT_FETCH_FAILED",
177:                 httpRequests = fetched.requestsIssued, fulltextObtained = false,
178:                 downloadFailureCategory = FULLTEXT_FAILURE_TIMEOUT
179:             )
180:             is XmlFetchResult.Failed -> return EmailExtractionOutcome(
181:                 emptyList(), emailExtractionMethod, "FULLTEXT_FETCH_FAILED",
182:                 httpRequests = fetched.requestsIssued, fulltextObtained = false
183:             )
184:         }
185: 
186:         return try {
187:             val emails = JatsXmlEmailParser.parse(xml)
188:             if (emails.isEmpty()) {
189:                 EmailExtractionOutcome(emptyList(), emailExtractionMethod, "NO_EMAIL_IN_FULLTEXT", httpRequests = 1)
190:             } else {
191:                 EmailExtractionOutcome(emails, emailExtractionMethod, null, httpRequests = 1)
192:             }
193:         } catch (e: Exception) {
194:             log.debug("Failed to parse JATS XML for {}: {}", pmcId, e.message)
195:             EmailExtractionOutcome(emptyList(), emailExtractionMethod, "XML_PARSE_FAILED", httpRequests = 1)
196:         }
197:     }
