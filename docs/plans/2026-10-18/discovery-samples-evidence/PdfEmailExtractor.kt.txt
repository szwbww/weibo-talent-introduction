59:         pdfUrl: String,
60:         knownAuthors: List<PaperAuthor>,
61:         sourceName: String,
62:         deadline: Instant? = null,
63:         onResponseHeaders: ((HttpHeaders) -> Unit)? = null
64:     ): EmailExtractionOutcome {
65:         val trafficBefore = DiscoveryTrafficMeter.currentScopeBytes()
66:         val uri = try {
67:             URI.create(pdfUrl)
68:         } catch (e: IllegalArgumentException) {
69:             log.debug("[{}] Invalid PDF URL: {}", sourceName, pdfUrl)
70:             return EmailExtractionOutcome(
71:                 emails = emptyList(),
72:                 methodUsed = "PDF_PARSE",
73:                 failureReason = "PDF_DOWNLOAD_FAILED",
74:                 httpRequests = 0,
75:                 fulltextObtained = false
76:             )
77:         }
78: 
79:         val downloaded = try {
80:             FetchRetry.retryOnRecoverableIo(
81:                 maxRetries = properties.maxRetries,
82:                 initialBackoffMs = properties.retryBackoffMs
83:             ) {
84:                 // 重试也不重新计时：共享 deadline 已过期就不再发新的请求。
85:                 if (deadlineExpired(deadline)) throw FulltextDeadlineExceededException(requestIssued = false)
86:                 downloadWithStreamLimit(uri, deadline, onResponseHeaders)
87:             }
88:         } catch (e: FulltextDeadlineExceededException) {
89:             log.debug("[{}] Fulltext download of {} exceeded the shared per-paper deadline", sourceName, pdfUrl)
90:             return EmailExtractionOutcome(
91:                 emptyList(), "PDF_PARSE", "PDF_DOWNLOAD_FAILED",
92:                 httpRequests = if (e.requestIssued) 1 else 0,
93:                 fulltextObtained = false,
94:                 downloadFailureCategory = FULLTEXT_FAILURE_TIMEOUT
95:             )
96:         } catch (e: PdfTooLargeException) {
97:             DiscoveryTrafficMeter.recordOversized(DiscoveryTrafficMeter.currentScopeBytes() - trafficBefore)
98:             log.debug("[{}] PDF {} too large", sourceName, pdfUrl)
99:             return EmailExtractionOutcome(
100:                 emptyList(), "PDF_PARSE", "PDF_TOO_LARGE", httpRequests = 1, fulltextObtained = false
101:             )
102:         } catch (e: Exception) {
103:             log.debug("[{}] Failed to download PDF {}: {}", sourceName, pdfUrl, e.message)
104:             return EmailExtractionOutcome(
105:                 emptyList(), "PDF_PARSE", "PDF_DOWNLOAD_FAILED",
106:                 httpRequests = 1,
107:                 fulltextObtained = false,
108:                 downloadFailureCategory = classifyDownloadFailure(e)
109:             )
110:         }
111: 
112:         return when (downloaded.kind) {
113:             ContentKind.PDF -> extractFromPdf(downloaded.bytes, knownAuthors, pdfUrl, sourceName, deadline)
114:             ContentKind.HTML -> extractFromHtml(downloaded.bytes, knownAuthors)
115:             ContentKind.OTHER -> EmailExtractionOutcome(
116:                 emptyList(), "PDF_PARSE", "PDF_DOWNLOAD_FAILED",
117:                 httpRequests = 1,
118:                 fulltextObtained = false,
119:                 downloadFailureCategory = FULLTEXT_FAILURE_INVALID_CONTENT
120:             )
121:         }

161:     private fun extractFromHtml(bytes: ByteArray, knownAuthors: List<PaperAuthor>): EmailExtractionOutcome {
162:         val html = String(bytes, StandardCharsets.UTF_8)
163:         if (anubisScript.containsMatchIn(html) && anubisTitle.containsMatchIn(html)) {
164:             return invalidHtml()
165:         }
166:         val (visible, emails) = SourceAuthorEmailResolver.resolveHtmlWithVisibility(
167:             html, knownAuthors, properties.blacklistPrefixes)
168:         if (!visible) return invalidHtml()
169:         if (emails.isEmpty()) {
170:             return EmailExtractionOutcome(
171:                 emptyList(), "HTML_FALLBACK", "NO_EMAIL_IN_HTML", httpRequests = 1, fulltextObtained = true
172:             )
173:         }
174:         return EmailExtractionOutcome(emails, "HTML_FALLBACK", null, httpRequests = 1, fulltextObtained = true)
175:     }

224:         deadline: Instant?,
225:         onResponseHeaders: ((HttpHeaders) -> Unit)?
226:     ): DownloadedContent {
227:         // R-1（V-4）：连接与响应头读取都在同一个剩余预算内 —— 有界 client 的连接/读取超时都取
228:         // min(既有配置, 剩余预算)，响应体读取仍由下面的分片 deadline 检查把关，两处都不重新计时。
229:         // R-1（V-4）：dispatch 前再判一次 —— 预算已尽就不发这次请求（0 请求，不是「发出去再超时」）。
230:         if (com.weibo.talentintroduction.config.BoundedFulltextHttp.remainingMsOrUnbounded(deadline) <= 0L) {
231:             throw FulltextDeadlineExceededException(requestIssued = false)
232:         }
233:         return boundedHttp.execute<DownloadedContent>(
234:             restTemplate, uri,
235:             PDF_DOWNLOAD_CONNECT_TIMEOUT_MS, properties.downloadTimeoutMs, deadline
236:         ) { response ->
237:             onResponseHeaders?.invoke(response.headers)
238:             val contentType = response.headers.contentType
239:             val isPdfContentType = contentType != null &&
240:                 (contentType.isCompatibleWith(MediaType.APPLICATION_PDF) ||
241:                  contentType.subtype?.lowercase() == "pdf")
242: 
243:             val maxSize = properties.maxPdfSizeBytes
244:             val buffer = java.io.ByteArrayOutputStream()
245:             val chunk = ByteArray(8192)
246:             var totalRead = 0L
247: 
248:             response.body.use { input ->
249:                 while (true) {
250:                     // R-1（V-4）：每次分片读取前后都对照绝对时限 —— 服务端只要在单次读超时前吐一点数据
251:                     // 就能让「按次读超时」永远不到期，因此这里必须按绝对 deadline 截断。
252:                     if (deadlineExpired(deadline)) throw FulltextDeadlineExceededException(requestIssued = true)
253:                     val n = input.read(chunk)
254:                     if (n == -1) break
255:                     totalRead += n
256:                     if (totalRead > maxSize) {
257:                         throw PdfTooLargeException()
258:                     }
259:                     buffer.write(chunk, 0, n)
260:                     if (deadlineExpired(deadline)) throw FulltextDeadlineExceededException(requestIssued = true)
261:                 }
262:             }
263: 
264:             val bytes = buffer.toByteArray()
265:             if (bytes.isEmpty()) throw RuntimeException("Empty body")
266: 
267:             val hasMagic = bytes.size >= 4 && bytes.take(4).toByteArray().contentEquals(magicBytes)
268:             if (isPdfContentType || hasMagic) {
269:                 return@execute DownloadedContent(bytes, ContentKind.PDF)
270:             }
271: 
272:             if (properties.htmlFallbackEnabled && isHtmlContent(contentType, bytes)) {
273:                 return@execute DownloadedContent(bytes, ContentKind.HTML)
274:             }
275: 
276:             DownloadedContent(bytes, ContentKind.OTHER)
277:         } ?: throw RuntimeException("Empty response")
278:     }

280:     private fun isHtmlContent(contentType: MediaType?, bytes: ByteArray): Boolean {
281:         if (contentType != null && contentType.isCompatibleWith(MediaType.TEXT_HTML)) {
282:             return true
283:         }
284:         val prefix = String(bytes, 0, minOf(bytes.size, 512), StandardCharsets.UTF_8)
285:             .trimStart()
286:             .lowercase()
287:         return prefix.startsWith("<!doctype html") || prefix.startsWith("<html")
288:     }

293:     private fun extractEmailsFromBytes(bytes: ByteArray, knownAuthors: List<PaperAuthor>, deadline: Instant?): List<AuthorEmail> {
294:         ByteArrayInputStream(bytes).use { stream ->
295:             PDDocument.load(stream).use { doc ->
296:                 val text = StringBuilder()
297:                 val clues = linkedSetOf<String>()
298:                 val contacts = PdfAuthorContactLayout.collect(
299:                     doc, properties.maxPages, knownAuthors, properties.tailPages,
300:                     beforePage = {
301:                         if (deadlineExpired(deadline)) throw FulltextDeadlineExceededException(requestIssued = true)
302:                     },
303:                     onPage = { page, pageText ->
304:                         if (page <= properties.maxPages) text.append(pageText).append('\n')
305:                         else clues += plainTextExtractor.extract(pageText, properties.blacklistPrefixes)
306:                         for (annotation in doc.getPage(page - 1).annotations) {
307:                             val uri = ((annotation as? PDAnnotationLink)?.action as? PDActionURI)?.uri ?: continue
308:                             if (!uri.startsWith("mailto:", ignoreCase = true)) continue
309:                             val address = uri.substring(7).substringBefore('?').trim()
310:                             if (!mailboxUri.matches(address)) continue
311:                             if (plainTextExtractor.extract(address, properties.blacklistPrefixes).singleOrNull() ==
312:                                 address.lowercase()) clues += address.lowercase()
313:                         }
314:                     }
315:                 )
316:                 return SourceAuthorEmailResolver.resolvePdf(
317:                     text.append('\n').append(clues.joinToString("\n")).toString(),
318:                     knownAuthors, contacts, properties.blacklistPrefixes
319:                 )
320:             }
321:         }
322:     }
