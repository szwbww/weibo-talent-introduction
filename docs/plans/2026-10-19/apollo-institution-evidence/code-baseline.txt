130:     private fun reportingWhere(filter: VerificationReportFilter): Pair<String, List<Any>> {
131:         val clauses = mutableListOf("i.owner_execution_id=?")
132:         val args = mutableListOf<Any>(filter.executionId)
133:         when (filter.route) {
134:             "APOLLO" -> clauses += "i.apollo_entered_at IS NOT NULL"
135:             "DIRECT" -> { clauses += "i.route=?"; args += filter.route }
136:         }
137:         when (filter.result) {
138:             "CLEAR" -> clauses += "i.outcome IN ('DIRECT_CLEAR','APOLLO_CLEAR')"
139:             "UNCLEAR" -> clauses += "i.outcome='APOLLO_UNCLEAR'"
140:             "PENDING" -> clauses += "i.outcome='PENDING'"
141:             "ERROR" -> clauses += "i.state='ERROR'"
142:             "QUEUED" -> clauses += "i.state IN ('QUEUED','RUNNING','RETRY_WAIT')"
143:             "PROTECTED" -> clauses += "i.outcome='PROTECTED'"
144:         }
145:         filter.batchNumber?.let { clauses += "i.batch_number=?"; args += it }
146:         if (filter.q.isNotEmpty()) {
147:             // Use ! as SQL escape, escaping literal !, %, _, and backslash explicitly.
148:             val q = "%" + filter.q.replace("!", "!!").replace("\\", "!\\").replace("%", "!%").replace("_", "!_") + "%"
149:             clauses += """(i.expert_doc_id LIKE ? ESCAPE '!' OR
150:                 CONCAT_WS(' ',JSON_UNQUOTE(JSON_EXTRACT(i.snapshot_json,'$.profile.givenNames')),
151:                     JSON_UNQUOTE(JSON_EXTRACT(i.snapshot_json,'$.profile.familyNames'))) LIKE ? ESCAPE '!' OR
152:                 JSON_UNQUOTE(JSON_EXTRACT(i.snapshot_json,'$.profile.email')) LIKE ? ESCAPE '!' OR
153:                 JSON_UNQUOTE(JSON_EXTRACT(i.snapshot_json,'$.profile.institution')) LIKE ? ESCAPE '!')"""
154:             repeat(4) { args += q }
155:         }
156:         return clauses.joinToString(" AND ") to args
157:     }

245:     @Transactional
246:     fun claimDue(now: Instant, leaseSeconds: Long): VerificationItem? {
247:         require(leaseSeconds > 0)
248:         val candidates = jdbc.query("""SELECT * FROM discovery_verification_item WHERE
249:             state='QUEUED' OR (state='RETRY_WAIT' AND next_attempt_at<=?) OR (state='RUNNING' AND lease_until<=?) ORDER BY id LIMIT 100""",
250:             itemMapper, now.verificationUtc(), now.verificationUtc())
251:         // One run per transaction, even when all of its candidates lost a race.
252:         val owner = candidates.firstOrNull()?.ownerExecutionId ?: return null
253:         lockRun(owner)
254:         for (candidate in candidates.filter { it.ownerExecutionId == owner }) {
255:             val item = lockedItem(candidate.id)
256:             if (!(item.state == VerificationState.QUEUED ||
257:                     item.state == VerificationState.RETRY_WAIT && item.nextAttemptAt?.let { it <= now } == true ||
258:                     item.state == VerificationState.RUNNING && item.leaseUntil?.let { it <= now } == true)) continue
259:             jdbc.update("UPDATE discovery_verification_item SET state='RUNNING',lease_token=?,generation=generation+1,lease_until=?,next_attempt_at=NULL,updated_at=? WHERE id=?",
260:                 UUID.randomUUID().toString(), now.plusSeconds(leaseSeconds).verificationUtc(), now.verificationUtc(), item.id)
261:             bump(owner, now)
262:             return findItem(item.id)
263:         }
264:         return null
265:     }

278:     @Transactional
279:     fun reserveAttemptWithBudget(itemId: Long, token: String, generation: Long, request: VerificationAttemptRequest,
280:                                  requestsPerMinute: Int, requestsPerDay: Int, maxRequestsPerItem: Int, now: Instant): VerificationAttemptReservation {
281:         require(requestsPerMinute > 0 && requestsPerDay > 0 && maxRequestsPerItem > 0)
282:         val item = lockItem(itemId)
283:         if (!owns(item, token, generation, now)) return VerificationAttemptReservation(deferred = VerificationBudgetReason.LEASE_LOST)
284:         // Lock is retained through COMMIT, not merely through INSERT. Same bound JDBC connection.
285:         if (request.provider == VerificationProvider.APOLLO) {
286:             val acquired = jdbc.execute(ConnectionCallback<Boolean> { connection ->
287:                 connection.prepareStatement("SELECT GET_LOCK('discovery_verification_apollo_budget',1)").use { statement ->
288:                     statement.executeQuery().use { rs -> rs.next(); rs.getInt(1) == 1 }
289:                 }.also { acquired ->
290:                     if (acquired) TransactionSynchronizationManager.registerSynchronization(object : TransactionSynchronization {
291:                         override fun afterCompletion(status: Int) {
292:                             connection.prepareStatement("SELECT RELEASE_LOCK('discovery_verification_apollo_budget')").use { it.execute() }
293:                         }
294:                     })
295:                 }
296:             }) == true
297:             if (!acquired) return VerificationAttemptReservation(deferred = VerificationBudgetReason.LOCK_BUSY)
298:         }
299:         val duplicate = jdbc.queryForObject("SELECT COUNT(*) FROM discovery_verification_attempt WHERE request_key=? FOR UPDATE", Long::class.java, request.requestKey)!!
300:         if (duplicate > 0) return VerificationAttemptReservation(deferred = VerificationBudgetReason.DUPLICATE_REQUEST)
301:         if (item.requestCount >= maxRequestsPerItem) return VerificationAttemptReservation(deferred = VerificationBudgetReason.ITEM_LIMIT)
302:         if (request.provider == VerificationProvider.APOLLO) {
303:             // Current reads are required even inside an outer REPEATABLE_READ transaction:
304:             // acquiring a named lock does not refresh an earlier MVCC snapshot.
305:             val start = LocalDate.ofInstant(now, ZoneOffset.UTC).atStartOfDay()
306:             val day = jdbc.queryForObject("SELECT COUNT(*) FROM discovery_verification_attempt WHERE provider='APOLLO' AND requested_at>=? AND requested_at<? FOR UPDATE",
307:                 Long::class.java, start, start.plusDays(1))!!
308:             if (day >= requestsPerDay) return VerificationAttemptReservation(deferred = VerificationBudgetReason.DAY_LIMIT)
309:             val minute = jdbc.queryForObject("SELECT COUNT(*) FROM discovery_verification_attempt WHERE provider='APOLLO' AND requested_at>? AND requested_at<=? FOR UPDATE",
310:                 Long::class.java, now.minusSeconds(60).verificationUtc(), now.verificationUtc())!!
311:             if (minute >= requestsPerMinute) return VerificationAttemptReservation(deferred = VerificationBudgetReason.MINUTE_LIMIT)
312:         }
313:         jdbc.update("""INSERT INTO discovery_verification_attempt (item_id,request_key,provider,query_kind,page_number,attempt_number,requested_at,result_code,request_json)
314:             VALUES (?,?,?,?,?,?,?,'DISPATCH_RESERVED',?)""", itemId, request.requestKey, request.provider.name, request.queryKind.name,
315:             request.pageNumber, request.attemptNumber, now.verificationUtc(), request.requestJson)
316:         val id = jdbc.queryForObject("SELECT id FROM discovery_verification_attempt WHERE request_key=?", Long::class.java, request.requestKey)!!
317:         jdbc.update("UPDATE discovery_verification_item SET request_count=request_count+1,updated_at=? WHERE id=?", now.verificationUtc(), itemId)
318:         bump(item.ownerExecutionId, now)
319:         return VerificationAttemptReservation(attemptId = id)
320:     }
